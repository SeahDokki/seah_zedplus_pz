VERSION = 1,

/*
 * ZED+ sandbox options.
 *
 * Option names map to SandboxVars.SZedPlus.<Name> in Lua. Read them through
 * SZedPlus.Config, never directly: the config layer applies fallbacks so the
 * mod still runs if an option is missing (older save, server override).
 *
 * Translation keys live in media/lua/shared/Translate/<LANG>/Sandbox.json
 * as Sandbox_<translation> and Sandbox_<translation>_tooltip. JSON, and no
 * language suffix in the filename - the folder carries the language.
 */
