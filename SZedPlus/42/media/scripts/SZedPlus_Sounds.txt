module SZedPlus
{
    /*
     * The Thriller easter egg's loop.
     *
     * Build 42 ships no sounds.txt of its own - the game's audio lives in FMOD
     * banks - but ScriptManager still parses `sound` blocks into
     * GameSoundScript, and a GameSound declared here is playable by name from
     * Lua through any emitter. That is the route a mod has.
     *
     * category and master are enum names, not free text: SoundCategory accepts
     * General, Item, Player, Zombie, World, Vehicle, Animals, Music, Meta,
     * Object and Auto; GameSound.MasterVolume accepts Primary, Ambient, Music
     * and VehicleEngine. Ambient puts it under the ambient volume slider, which
     * is where a background loop belongs - a player who has turned ambience
     * down has said what they want.
     *
     * The name is prefixed because sound names are one flat global namespace
     * shared with every other mod, the same as texture names.
     */
    sound SZedPlus_Thriller
    {
        category = World,
        master = Ambient,
        loop = true,

        /* One at a time per emitter: the lead is the only emitter, but a
           second copy starting over the first would be audible and wrong. */
        maxInstancesPerEmitter = 1,

        clip
        {
            file = media/sound/mj_thriller_egg.ogg,

            /* Fifty tiles, not the fifteen this started at.
               Fifteen was reasoned from "roughly where the ring becomes
               visible", which sounded right and was wrong in play: the ring is
               placed seven to twelve tiles away and the dance breaks up at six,
               so the music only existed inside the window where the player was
               already on top of it. The point of the sound is to be heard from
               somewhere else and walked towards. Below two tiles it stops
               getting louder, so standing in the middle is not painful. */
            distanceMin = 2,
            distanceMax = 50,
            volume = 0.5,
        }
    }

    /*
     * The Witch's scream, as four takes.
     *
     * One GameSound with several clips is how variation is done: GameSound has
     * getRandomClip and picks per play, so four clips here is four takes of the
     * same scream rather than four sounds to choose between in Lua. Hearing the
     * identical scream twice in one night would undo her.
     *
     * distanceMax matches the Witch's screamRadius of 30 in SZedPlus_Forms, on
     * purpose: the player hears her from exactly as far as the zombies she is
     * calling do. If that radius is retuned, this follows it.
     *
     * Primary rather than Ambient - this is a creature, not weather.
     */
    sound SZedPlus_WitchScream
    {
        category = Zombie,
        master = Primary,
        loop = false,
        maxInstancesPerEmitter = 1,

        clip
        {
            file = media/sound/witch_screetch_1.ogg,
            distanceMin = 3,
            distanceMax = 30,
            volume = 1.0,
        }

        clip
        {
            file = media/sound/witch_screetch_2.ogg,
            distanceMin = 3,
            distanceMax = 30,
            volume = 1.0,
        }

        clip
        {
            file = media/sound/witch_screetch_3.ogg,
            distanceMin = 3,
            distanceMax = 30,
            volume = 1.0,
        }

        clip
        {
            file = media/sound/witch_screetch_4.ogg,
            distanceMin = 3,
            distanceMax = 30,
            volume = 1.0,
        }
    }
}
